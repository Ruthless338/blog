1.  单调栈：维护一个单调递增或单调递减的栈
    常用于：下一个更大元素；下一个更小元素；左边第一个更大/更小元素；柱状图最大矩形（求当前柱子左右更矮的柱子）；每日温度
    ```cpp
    // a[1..n]
    // ans[i] = i右边第一个比a[i]大的位置，不存在则为-1
    stack<int> st;

    for(int i=n;i>=1;i--) {
        // 保持栈中对应元素单调递减
        while(!st.empty() && a[st.top()]<=a[i])
            st.pop();

        if(st.empty()) ans[i]=-1;
        else ans[i]=st.top();

        st.push(i);
    }
    ```

2. 单调队列：一般用deque实现，用于维护窗口最大值/最小值
    以窗口最大值为例，保持队列单调递减（当前元素大于队尾元素时弹出队尾并插入当前元素），队首永远是窗口最大值，每个元素最多进队一次、出队一次，所以总复杂度 O(n)
    ```cpp
    // a[1..n]
    deque<int> q;

    for(int i=1;i<=n;i++) {
        // 1. 删除已经离开窗口的元素
        while(!q.empty() && q.front() < i-k+1)
            q.pop_front();

        // 2. 保持队列对应元素单调递减
        while(!q.empty() && a[q.back()] <= a[i])
            q.pop_back();

        q.push_back(i);

        // 3. 队首就是当前窗口最大值
        if(i>=k)
            cout << a[q.front()] << " ";
    }
    ```

    vector<int> g[N];

    int lca(int u, int fa, int p, int q) {
        if(u==p || u==q) return u;

        int cnt=0, res=0;

        for(int v:g[u]) {
            if(v==fa) continue;

    ll a[N]; // a[1]->[1,1]  a[2]->[1,2] a[3]->[3,3] a[4]->[1,4] a[6]->[5,6] a[8]->[1,8]
    int lowbit(int x) { // 取出 x 的最低一位1所代表的数 如lowbit(12)=4
        return x&(-x);
    }
    void add(int x,int k) {  // 沿着 x, x+lowbit(x), ... 往上更新（找所有包含 x 的区间）
        while(x<=n) {
            a[x]+=k;
            x+=lowbit(x);
        }
    }

        while(x>0) {
            res+=a[x];
            x-=lowbit(x);
        }
        return res;
    } //[l,r] 区间和：query(r)-query(l-1)

    重要应用：归并排序求逆序对

    ```cpp
    void mergeSort(vector<int>& a, int l, int r, vector<int>& tmp) {
        if (l >= r) return;

        int mid = l + (r - l) / 2;

        mergeSort(a, l, mid, tmp);
        mergeSort(a, mid + 1, r, tmp);

        int i = l;
        int j = mid + 1;
        int k = l;

        while (i <= mid && j <= r) {
            if (a[i] <= a[j]) tmp[k++] = a[i++];
            else tmp[k++] = a[j++];
        }

        while (i <= mid) tmp[k++] = a[i++];
        while (j <= r) tmp[k++] = a[j++];

        for (int p = l; p <= r; p++) {
            a[p] = tmp[p];
        }
    }
    ```

    void quickSort(vector<int>& nums,int l,int r)
    {
        if(l>=r)
            return;

        int pivot=nums[r];

        int i=l;

        for(int j=l;j<r;j++)
        {
            if(nums[j]<pivot)
            {
                swap(nums[i],nums[j]);
                i++;
            }
        }
        swap(nums[i],nums[r]);//最后交换标杆元素

        quickSort(nums,l,i-1);
        quickSort(nums,i+1,r);
    }

11. 二分查找
    lower_bound：找第一个大于等于 target 的位置

    ```cpp
    int lowerBound(vector<int>& nums, int target) {
        int l = 0;
        int r = nums.size();
        while (l < r) {
            int mid = l + (r - l) / 2;
            if (nums[mid] >= target) { //将等号去掉即为upper_bound：找第一个大于target的位置
                r = mid;
            } else {
                l = mid + 1;
            }
        }
        return l;
    }
    ```

32. 贪心
    求到达最后位置的最少跳跃次数->不需要用动态规划，O(n)就够了

    ```cpp
    int jump(vector<int>& nums) {
        int end = 0;// 当前这一步能覆盖的边界
        int farthest = 0;//下一步能覆盖的边界
        int steps = 0;//步数

        for (int i = 0; i < nums.size() - 1; i++) {
            farthest = max(farthest, i + nums[i]);

            if (i == end) {
                steps++;
                end = farthest;
            }
        }

        return steps;
    }
    ```

    3. 最长递增子序列LIS：
    解法1. O(n^2)，dp[i]表示以nums[i]结尾的最长递增子序列长度
    ```cpp
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> dp(n, 1);
        int ans = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (nums[j] < nums[i]) {
                    dp[i] = max(dp[i], dp[j] + 1);
                }
            }
            ans = max(ans, dp[i]);
        }
        return ans;
    }
    ```

    解法2. O(nlogn)，贪心+二分，tails[i]表示长度为i+1的递增子序列中，最小的结尾元素，整个tails构成了最优的递增子序列

    ```cpp
    int lengthOfLIS(vector<int>& nums)
    {
        vector<int> tails;
        for(int x:nums)
        {
            auto it=lower_bound(tails.begin(), tails.end(), x);
            if(it==tails.end())
                tails.push_back(x);
            else
                *it=x;
        }
        return tails.size();
    }
    ```

5. 区间 DP：石子合并

    有一排石子，每次只能合并相邻两堆，代价为两堆石子总数，求全部合并的最小代价。

    定义：$dp[l][r]$ = 将区间 $[l,r]$ 合并成一堆的最小代价

    最后一次合并一定把区间分成 $[l,k]$ 和 $[k+1,r]$

    因此：

    $$dp[l][r] = \min_{l \le k < r} \{dp[l][k] + dp[k+1][r] + sum(l,r)\}$$
    代码框架：

    ```cpp
    vector<vector<long long>> dp(n, vector<long long>(n));
    vector<long long> pre(n + 1);

    for (int i = 0; i < n; i++) {
        pre[i + 1] = pre[i] + a[i];
    }

    for (int len = 2; len <= n; len++) {
        for (int l = 0; l + len - 1 < n; l++) {
            int r = l + len - 1;
            dp[l][r] = LLONG_MAX;
            long long sum = pre[r + 1] - pre[l];
            for (int k = l; k < r; k++) {
                dp[l][r] = min(
                    dp[l][r],
                    dp[l][k] + dp[k + 1][r] + sum
                );
            }
        }
    }
    ```

    复杂度：$O(n^3)$

    区间 DP 的识别信号：

    对一个区间不断分割或合并；
    最后一步枚举分割点；
    状态通常为 dp[l][r]。

6. 戳气球：枚举最后一个

    这类题最关键的思想是：

    “第一个被选择的元素”周围情况会变化，但“最后一个被选择的元素”两侧边界已经确定。

    定义：$dp[l][r]$ = 只处理开区间 $(l,r)$ 内气球的最大收益

    若 $k$ 是最后一个被戳的气球：

    $$dp[l][r] = \max_{l < k < r} [dp[l][k] + dp[k][r] + a_l \cdot a_k \cdot a_r]$$
    这是非常经典的”换一个角度定义最后一步”。

7. 状态压缩 DP：旅行商模型
    当 $n \le 20$，需要记录一个集合“哪些选过/访问过”，可以用二进制集合表示已访问节点。
    定义：$dp[mask][i]$ = 已经访问集合 $mask$，最后位于 $i$ 的最短距离
    转移：
    $$dp[mask \cup \{j\}][j] = \min_{i \in mask} [dp[mask][i] + w_{ij}]$$
    其中 $i \in mask, j \notin mask$,    复杂度：$O(2^n \cdot n^2)$
    ```cpp
    int total = 1 << n;
    vector<vector<long long>> dp(
        total, vector<long long>(n, INF)
    );
    dp[1 << 0][0] = 0;
    for (int mask = 0; mask < total; mask++) {
        for (int u = 0; u < n; u++) {
            if (!(mask & (1 << u))) continue;
            if (dp[mask][u] == INF) continue;
            for (int v = 0; v < n; v++) {
                if (mask & (1 << v)) continue;
                int nextMask = mask | (1 << v);
                dp[nextMask][v] = min(
                    dp[nextMask][v],
                    dp[mask][u] + w[u][v]
                );
            }
        }
    }
    ll ans = INF;
    int full = (1 << n) - 1;
    for(int i=0;i<n;i++)
        ans = min(ans, dp[full][i]); //从 0 出发，访问所有点，最后停在任意点
    cout << ans << '\n';
    ```
    有 n 个人和 n 个任务，cost[i][j] 表示第 i 个人做第 j 个任务的代价，每个任务只能分配一次，求最小总代价。
    ```cpp
    int total = 1 << n;
    vector<ll> dp(total, INF);

    dp[0] = 0;

    for(int mask=0; mask<total; mask++) {
        int i = __builtin_popcount(mask);  // 用来计算一个整数的二进制表示里有多少个1, i代表当前轮到第i个人（从0开始）

        for(int j=0; j<n; j++) {
            if(mask & (1<<j)) continue;     // 任务j已经被选了

            int nextMask = mask | (1<<j);

            dp[nextMask] = min(
                dp[nextMask],
                dp[mask] + cost[i][j]
            );
        }
    }

    cout << dp[(1<<n)-1] << '\n';
    ```

24. DAG 上的动态规划

    有向无环图的最长路径可以按拓扑序 DP。

    定义：$dp[v]$ = 到达 $v$ 的最长路径

    对于边 $u \to v$，转移：

    $$dp[v] = \max(dp[v], dp[u] + w(u,v))$$
    为什么必须按拓扑序？因为处理 $v$ 前，需要确保所有可能转移到 $v$ 的前驱都已经处理。

    DAG 上即使存在负权边，也能在线性时间内求最短路径：$O(V+E)$

二、树形 DP 经典题

1. 树上最大独立集：选择若干节点，要求相邻节点不能同时选择，最大化权值和。
    定义：
    - $dp[u][0]$ = 不选 $u$ 时，子树最大权值
    - $dp[u][1]$ = 选择 $u$ 时，子树最大权值
    转移：
    $$dp[u][1] = w_u + \sum_{v \in child(u)} dp[v][0]$$
    因为选了 $u$，孩子不能选。
    $$dp[u][0] = \sum_{v \in child(u)} \max(dp[v][0], dp[v][1])$$
    代码：
    ```cpp
    vector<array<long long, 2>> dp;
    vector<vector<int>> graph;
    vector<int> weight;

    void dfs(int u, int parent) {
        dp[u][0] = 0;
        dp[u][1] = weight[u];
        for (int v : graph[u]) {
            if (v == parent) continue;
            dfs(v, u);
            dp[u][1] += dp[v][0];
            dp[u][0] += max(dp[v][0], dp[v][1]);
        }
    }
    ```

    最终答案：$\max(dp[root][0], dp[root][1])$

2. 树的直径

    树的直径是任意两点之间的最长路径。
    定义：
    down[u]=从 u 向其子树走的最长路径
    对于节点 u，经过它的最长路径由最大的两条子树路径组成。
    ```cpp
    vector<int> g[N];
    int ans = 0;

    int dfs(int u, int fa) {
        int mx1 = 0, mx2 = 0;  // 最长、次长向下路径
        for(int v : g[u]) {
            if(v == fa) continue;
            int d = dfs(v, u) + 1;
            if(d > mx1) {
                mx2 = mx1;
                mx1 = d;
            }
            else if(d > mx2) {
                mx2 = d;
            }
        }
        ans = max(ans, mx1 + mx2);
        return mx1;  // 向父节点只能贡献最长的一条
    }
    ```

3. 树最大路径和

    节点权值可能为负，路径可以从任意节点开始和结束。
    代码：

    ```cpp
    vector<int> g[N];
    ll w[N];
    ll ans = LLONG_MIN;

    ll dfs(int u, int fa) {
        ll mx1 = 0, mx2 = 0;

        for(int v : g[u]) {
            if(v == fa) continue;

            ll x = max(0LL, dfs(v, u));

            if(x > mx1) {
                mx2 = mx1;
                mx1 = x;
            }
            else if(x > mx2) {
                mx2 = x;
            }
        }

        // 完整路径可以从一个子树经过u，再进入另一个子树
        ans = max(ans, w[u] + mx1 + mx2);

        // 向父节点只能贡献一条路径
        return w[u] + mx1;
    }
    ```

4. 树上背包

    题目形式：
    每个节点有价值；选择节点数量有限；选择关系受树结构约束；求最大价值。

    定义：$dp[u][j]$ = 在 $u$ 的子树中选择 $j$ 个节点的最大价值

    初始化：$dp[u][1] = w_u$

    合并一个孩子 $v$ 时：

    $$dp[u][j+k] = \max[dp[u][j+k], dp[u][j] + dp[v][k]]$$
    这相当于每个子树都是一个“物品组”，需要枚举给该子树分配多少选择数量。

    骨架：

    ```cpp
    void dfs(int u, int parent) {
        size[u] = 1;
        dp[u][1] = value[u];

        for (int v : graph[u]) {
            if (v == parent) continue;

            dfs(v, u);

            for (int j = min(size[u], m); j >= 1; j--) {
                for (int k = 1;
                    k <= size[v] && j + k <= m;
                    k++) {
                    dp[u][j + k] = max(
                        dp[u][j + k],
                        dp[u][j] + dp[v][k]
                    );
                }
            }

            size[u] += size[v];
        }
    }
    ```

    识别信号：

    树上选择固定数量节点；
    每个子树需要分配容量；
    合并子树时出现两层容量循环。

5. 换根 DP：每个节点到其他节点的距离和

    要求对每个节点 $u$，计算：

    $$ans[u] = \sum_v dist(u,v)$$
    第一遍 DFS

    计算：
    - $size[u]$ = $u$ 子树节点数
    - $down[u]$ = $u$ 到子树所有节点的距离和

    转移：

    $$size[u] = 1 + \sum_v size[v]$$

    $$down[u] = \sum_v (down[v] + size[v])$$
    因为从 $u$ 到 $v$ 子树的每个节点，都比从 $v$ 出发多走一条边。

    第二遍 DFS

    已知 $ans[u]$，将根从 $u$ 移到孩子 $v$：

    - $v$ 子树中的 $size[v]$ 个节点距离全部减少 1
    - 其余 $n - size[v]$ 个节点距离全部增加 1

    所以：

    $$ans[v] = ans[u] + n - 2 \cdot size[v]$$
    这是换根 DP 最经典的公式。

    int dijkstra() {
        dis[s]=0;
        ty temp={s,dis[s]};
        q.push(temp);
        while(!q.empty()) {
            temp=q.top();
            q.pop();
            if(vis[temp.x]) continue;
            vis[temp.x]=true;
            for(auto [v,w] : e[temp.x]) {
                if(vis[v]) continue;
                if(dis[v]>dis[temp.x]+w) {
                    dis[v]=dis[temp.x]+w;
                    q.push({v,dis[v]});
                }
            }
        }

    }

38. 差分约束

    如果题目出现：

    $$
    x_i-x_j\le c
    $$

    移项：

    $$
    x_i\le x_j+c
    $$

    对应一条最短路边：

    ```text
    j -> i，权值 c
    ```

    因为最短路松弛：

    $$
    dis[i]\le dis[j]+c
    $$

    所以：

    $$
    \boxed{x_i-x_j\le c\Rightarrow j\rightarrow i(c)}
    $$

    其他形式：

    $$
    x_i-x_j\ge c
    $$

    变成：

    $$
    x_j-x_i\le -c
    $$

    所以：

    ```text
    i -> j，权值 -c
    ```

    相等：

    $$
    x_i-x_j=c
    $$

    拆成两个不等式。

46. 2-SAT

    每个变量只有：

    ```text
    选 / 不选
    true / false
    ```

    而条件类似：

    ```text
    A 和 B 不能同时选
    A 和 B 至少选一个
    如果选 A，则必须选 B
    ```

    可以转换成蕴含关系。

    1. A 和 B 不能同时成立
    $$ \neg(A\land B) $$

    等价于：

    $$ A\Rightarrow \neg B $$

    和：

    $$ B\Rightarrow \neg A $$

    建：

    A -> notB
    B -> notA
    2. A 和 B 至少一个成立
    $$ A\lor B $$

    等价于：

    notA -> B
    notB -> A
    3. A 成立，则 B 必须成立
    $$ A\Rightarrow B $$

    建：

    A -> B
    notB -> notA
    4. A 和 B 必须不同

    两种合法：

    A=true,  B=false
    A=false, B=true

    可以理解成：

    A -> notB
    B -> notA
    notA -> B
    notB -> A

    最后：

    > 如果 `x` 和 `¬x` 在同一个 SCC 中，则无解。

    ### 识别信号

47. 斜率优化 DP

    这个我建议你**理解，不要求自己推导所有题**。

    适合优化这种：

    $$
    dp[i]=\min_j\{dp[j]+f(i,j)\}
    $$

    暴力枚举：

    $$
    O(n^2)
    $$

    如果把式子展开之后，可以写成：

    $$
    Y_j=K_iX_j+B_i
    $$

    或者：

    $$
    k_jx_i+b_j
    $$

    那么每一个 `j` 可以看成一条直线：

    $$
    y=k_jx+b_j
    $$

    对于每一个 `i`：

    > 求所有直线在 \(x=x_i\) 时的最小/最大值。

    这就是：

    $$
    \boxed{\text{Convex Hull Trick / 斜率优化}}
    $$


    ## 经典例子

    假设：

    $$
    dp[i]
    =
    \min_{j<i}
    \left(
    dp[j]+(x_i-x_j)^2+C
    \right)
    $$

    展开：

    $$
    dp[i]
    =
    x_i^2+C
    +
    \min_j
    \left(
    dp[j]+x_j^2-2x_ix_j
    \right)
    $$

    对于固定 `j`：

    $$
    k_j=-2x_j
    $$

    $$
    b_j=dp[j]+x_j^2
    $$

    于是：

    $$
    dp[i]=x_i^2+C+\min_j(k_jx_i+b_j)
    $$

    变成：

    > 有很多条直线，询问 \(x_i\) 处的最小 y。
    ```cpp
    #include <bits/stdc++.h>
    using namespace std;

    typedef long long ll;
    typedef __int128 i128;

    struct Line {
        ll k, b;

        ll value(ll x) const {
            return k * x + b;
        }
    };

    deque<Line> q;

    // 判断中间直线 b 是否永远没有用
    // 这里要求斜率：a.k > b.k > c.k
    bool useless(Line a, Line b, Line c) {
        // intersection(a,b) >= intersection(b,c)
        return (i128)(b.b - a.b) * (b.k - c.k)
            >= (i128)(c.b - b.b) * (a.k - b.k);
    }

    // 加入一条新直线
    // 要求新加入的斜率单调不增
    void addLine(ll k, ll b) {

        // 斜率相同，只保留截距更小的那条
        if(!q.empty() && q.back().k == k) {
            if(q.back().b <= b)
                return;
            q.pop_back();
        }

        Line now = {k, b};

        while(q.size() >= 2 &&
            useless(q[q.size()-2], q[q.size()-1], now)) {
            q.pop_back();
        }

        q.push_back(now);
    }

    // 查询 x 处最小值
    // 要求查询 x 单调不减
    ll query(ll x) {
        while(q.size() >= 2 &&
            q[0].value(x) >= q[1].value(x)) {
            q.pop_front();
        }

        return q.front().value(x);
    }
    const ll INF = (1LL << 62);

    int n;
    ll C;
    ll x[100005];
    ll dp[100005];

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);

        cin >> n >> C;

        for(int i=1;i<=n;i++)
            cin >> x[i];

        // 假设 dp[1] = 0
        dp[1] = 0;

        // j=1 对应的直线
        addLine(
            -2 * x[1],
            dp[1] + x[1] * x[1]
        );

        for(int i=2;i<=n;i++) {

            dp[i] =
                x[i] * x[i]
                + C
                + query(x[i]);

            // i 以后可以作为 j
            addLine(
                -2 * x[i],
                dp[i] + x[i] * x[i]
            );
        }

        cout << dp[n] << '\n';
    }
    ```

    ### 识别信号

    看到：

    $$
    dp[i]=\min_j
    $$

    并且展开后出现：

    也就是：

    > 一部分只和 `j` 有关 × 一部分只和 `i` 有关。

            n = _n;
            m = _m;
            memset(a, 0, sizeof(a));
        }
    };

    Matrix operator * (const Matrix &A, const Matrix &B) {

        Matrix C(A.n, B.m);

        for(int i=0;i<A.n;i++) {
            for(int k=0;k<A.m;k++) {

                if(A.a[i][k] == 0)
                    continue;

                for(int j=0;j<B.m;j++) {

                    C.a[i][j] =
                        (C.a[i][j]
                        + A.a[i][k] * B.a[k][j] % MOD)
                        % MOD;
                }
            }
        }

        return C;
    }

    Matrix matrixPow(Matrix A, long long k) {

        Matrix res(A.n, A.n);

        // 单位矩阵
        for(int i=0;i<A.n;i++)
            res.a[i][i] = 1;

        while(k) {

            if(k & 1)
                res = res * A;

            A = A * A;

            k >>= 1;
        }

        return res;
    }
    ll fib(long long n) {

        if(n == 0)
            return 0;

        Matrix A(2,2);

        A.a[0][0] = 1;
        A.a[0][1] = 1;
        A.a[1][0] = 1;
        A.a[1][1] = 0;

        Matrix R = matrixPow(A, n-1);

        // [F1, F0] = [1, 0]
        return R.a[0][0];
    }
