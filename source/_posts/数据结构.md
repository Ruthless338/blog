1. KMP
首先维护一个next数组，含义为最长相等前后缀长度，如
p    = a b a b d
next = 0 0 1 2 0
对于主串
s: a b a b a b a b d
当j=4时，字符不等，KMP算法j = next[j - 1] = next[3] = 2
主串指针 i 不回退，模式串指针 j 根据 next 数组回退
构造next数组：
for (int i = 1, j = 0; i < p.size(); i++) {
    while (j > 0 && p[i] != p[j]) {
        j = next[j - 1];
    }

    if (p[i] == p[j]) {
        j++;
    }

    next[i] = j;
}
匹配主串：
for (int i = 0, j = 0; i < s.size(); i++) {
    while (j > 0 && s[i] != p[j]) {
        j = next[j - 1];
    }

    if (s[i] == p[j]) {
        j++;
    }

    if (j == p.size()) {
        return i - p.size() + 1;
    }
}
时间复杂度为O(n+m)
2. 反转链表-O(n)
ListNode* reverseList(ListNode* head) {
    ListNode* prev = nullptr;
    ListNode* cur = head;

    while (cur != nullptr) {
        ListNode* nxt = cur->next;
        cur->next = prev;
        prev = cur;
        cur = nxt;
    }

    return prev;
}

TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
    if (!root || root == p || root == q) return root;

    TreeNode* left = lowestCommonAncestor(root->left, p, q);
    TreeNode* right = lowestCommonAncestor(root->right, p, q);

    if (left && right) return root;
    if (left) return left;
    return right;
}
15. 验证BST（二叉搜索树）：BST 要求整棵左子树都小于 root，整棵右子树都大于 root
中序遍历递增：
TreeNode* prev = nullptr;

bool isValidBST(TreeNode* root) {
    if (!root) return true;

    if (!isValidBST(root->left)) return false;

    if (prev && prev->val >= root->val) return false;
    prev = root;

    return isValidBST(root->right);
}

22. 线段树
struct node{
	int l,r,sum,add;
}tr[N*4]; //l和r分别表示tr这个节点的左右边界，sum表示区间和，add是懒标记，用于记录其所有子节点需要加的值
void pushup(int p)  {//向上更新
	tr[p].sum=tr[p*2].sum+tr[p*2+1].sum;
}
void pushdown(int p) { //向下更新
	int lc=p*2,rc=p*2+1,add=tr[p].add;
	tr[lc].sum+=add*(tr[lc].r-tr[lc].l+1);
	tr[rc].sum+=add*(tr[rc].r-tr[rc].l+1);
	tr[lc].add+=add;
	tr[rc].add+=add;
	tr[p].add=0;
}
void build(int p,int l,int r) {
	tr[p]={l,r,a[l],0};
	if(l==r) return ;
	int mid=(l+r)>>1;
	build(p*2,l,mid); //左子树
	build(p*2+1,mid+1,r); //右子树
	pushup(p);
}
void update(int p,int l,int r,int k) {//p是当前节点，x是要求改的叶子结点，k是增加或减少值
	if(l<=tr[p].l&&r>=tr[p].r) 
	{
		tr[p].sum+=k*(tr[p].r-tr[p].l+1);
		tr[p].add+=k;
		return ;
	}
	int mid=(tr[p].l+tr[p].r)>>1;
	pushdown(p);
	if(l<=mid) update(p*2,l,r,k); //左子树
	if(r>mid) update(p*2+1,l,r,k);  //右子树
	pushup(p);
}
int query(int p,int l,int r) { //p是当前节点，l和r是查询区间
	if(l<=tr[p].l&&r>=tr[p].r) return tr[p].sum;
	pushdown(p);
	int mid=(tr[p].l+tr[p].r)>>1,sum=0;
	if(l<=mid) sum+=query(p*2,l,r);
	if(r>mid) sum+=query(p*2+1,l,r);
	return sum;
}

int lowbit(int x) { // 取出 x 的最低一位1所代表的数 如lowbit(12)=4
    return x&(-x);
}
void add(int x,int k) {  // 沿着 x, x+lowbit(x), ... 往上更新（找所有包含 x 的区间）
    while(x<=n) {
		a[x]+=k;
		x+=lowbit(x);
	}
}
ll query(int x) {  // 沿着 x, x-lowbit(x), ... 往回累加（把前缀拆成若干段）
	ll res=0;	
	while(x>0) {
		res+=a[x];
		x-=lowbit(x);
	}
	return res;
}

归并排序
void mergeSort(vector<int>& a, int l, int r, vector<int>& tmp) {
    if (l >= r) return;

    int mid = l + (r - l) / 2;

    mergeSort(a, l, mid, tmp);
    mergeSort(a, mid + 1, r, tmp);

    int i = l;
    int j = mid + 1;
    int k = l;

    while (i <= mid && j <= r) {
        if (a[i] <= a[j]) tmp[k++] = a[i++];
        else tmp[k++] = a[j++];
    }

    while (i <= mid) tmp[k++] = a[i++];
    while (j <= r) tmp[k++] = a[j++];

    for (int p = l; p <= r; p++) {
        a[p] = tmp[p];
    }
}
快速排序
void quickSort(vector<int>& nums,int l,int r)
{
    if(l>=r)
        return;

    int pivot=nums[r];

    int i=l;

    for(int j=l;j<r;j++)
    {
        if(nums[j]<pivot)
        {
            swap(nums[i],nums[j]);
            i++;
        }
    }

    swap(nums[i],nums[r]);//最后交换标杆元素

    quickSort(nums,l,i-1);
    quickSort(nums,i+1,r);
}
27. 二分查找
lower_bound：找第一个大于等于 target 的位置
int lowerBound(vector<int>& nums, int target) {
    int l = 0;
    int r = nums.size();

    while (l < r) {
        int mid = l + (r - l) / 2;

        if (nums[mid] >= target) { //将等号去掉即为upper_bound：找第一个大于target的位置
            r = mid;
        } else {
            l = mid + 1;
        }
    }

    return l;
}
